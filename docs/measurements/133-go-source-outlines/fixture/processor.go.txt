// Processor applies the pipeline's per-stage transformations.
// It is deliberately long: the interesting navigation problem in this
// repository is a file of this size, not a toy example.
type Processor struct {
	stage    string
	records  []string
	rejected int
}

// reconcile merges a batch after every attempt has settled. Its signature
// spans several lines on purpose.
func (p *Processor) reconcile(
	left []string,
	right []string,
	attempt int,
	force bool,
) ([]string, error) {
	if attempt <= 0 {
		return nil, fmt.Errorf("attempt must be positive")
	}
	out := make([]string, 0, len(left)+len(right))
	out = append(out, left...)
	for _, r := range right {
		if force || !strings.Contains(r, "duplicate") {
			out = append(out, r)
		}
	}
	return out, nil
}

// stats returns the processor's counters as a printable line.
func (p *Processor) stats() string {
	return fmt.Sprintf("stage=%s records=%d rejected=%d", p.stage, len(p.records), p.rejected)
}

// newProcessor builds a processor for one stage.
func newProcessor(stage string) *Processor {
	return &Processor{stage: stage}
}
